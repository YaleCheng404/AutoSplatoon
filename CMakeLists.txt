cmake_minimum_required(VERSION 3.25...4.4)
project(AutoSplatoon VERSION 1.0.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
include(CTest)
include(FetchContent)
include(GNUInstallDirs)
set(CMAKE_INSTALL_BINDIR ".")
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.4" CACHE STRING "Minimum macOS version")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.json" dependency_lock)
string(JSON qt_version GET "${dependency_lock}" qt version)
string(JSON firmware_hash GET "${dependency_lock}" firmware sha256)
find_package(Qt6 ${qt_version} EXACT REQUIRED COMPONENTS Widgets SerialPort Concurrent Svg)
qt_standard_project_setup()
foreach(asset IN ITEMS firmware font)
    string(JSON asset_hash GET "${dependency_lock}" "${asset}" sha256)
    if(asset STREQUAL "firmware")
        set(asset_file "${CMAKE_CURRENT_SOURCE_DIR}/firmware/PRO-UART0.bin")
    else()
        set(asset_file "${CMAKE_CURRENT_SOURCE_DIR}/AutoSplatoon/fonts/NotoSansCJKsc-Regular.otf")
    endif()
    file(SHA256 "${asset_file}" actual_hash)
    if(NOT actual_hash STREQUAL asset_hash)
        message(FATAL_ERROR "Bundled ${asset} checksum mismatch")
    endif()
endforeach()
string(JSON opencv_url GET "${dependency_lock}" opencv url)
string(JSON opencv_hash GET "${dependency_lock}" opencv sha256)
set(BUILD_LIST core,imgproc,photo CACHE STRING "" FORCE)
foreach(option IN ITEMS BUILD_TESTS BUILD_PERF_TESTS BUILD_EXAMPLES BUILD_opencv_apps
        BUILD_opencv_python3 BUILD_SHARED_LIBS WITH_IPP WITH_OPENCL WITH_ITT WITH_UNIFONT
        WITH_EIGEN WITH_LAPACK WITH_FFMPEG WITH_GSTREAMER WITH_OPENEXR WITH_OPENJPEG
        WITH_WEBP WITH_TIFF WITH_PNG WITH_JPEG OPENCV_GENERATE_PKGCONFIG)
    set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(CPU_DISPATCH "" CACHE STRING "" FORCE)
FetchContent_Declare(opencv URL "${opencv_url}" URL_HASH "SHA256=${opencv_hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_AUTORCC OFF)
FetchContent_MakeAvailable(opencv)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(OpenCV_LIBS opencv_core opencv_imgproc opencv_photo)
set(OpenCV_INCLUDE_DIRS "${OPENCV_CONFIG_FILE_INCLUDE_DIR}")
foreach(module IN ITEMS core flann geometry imgproc photo)
    set_target_properties(opencv_${module} PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    list(APPEND OpenCV_INCLUDE_DIRS "${opencv_SOURCE_DIR}/modules/${module}/include")
endforeach()
add_library(autosplatoon_core STATIC
    AutoSplatoon/drawingplanner.cpp AutoSplatoon/actionqueueplayer.cpp
    AutoSplatoon/imageprocessor.cpp AutoSplatoon/serialconnection.cpp
    AutoSplatoon/canvasview.cpp AutoSplatoon/flashdialog.cpp
    AutoSplatoon/mainwindow.cpp AutoSplatoon/manualcontroldialog.cpp AutoSplatoon/resources.qrc)
target_include_directories(autosplatoon_core PUBLIC AutoSplatoon ${OpenCV_INCLUDE_DIRS})
target_link_libraries(autosplatoon_core PUBLIC Qt6::Widgets Qt6::SerialPort Qt6::Concurrent Qt6::Svg ${OpenCV_LIBS})
target_compile_definitions(autosplatoon_core PRIVATE AUTOSPLATOON_FIRMWARE_SHA256="${firmware_hash}")
if(MSVC)
    target_compile_options(autosplatoon_core PUBLIC /utf-8 /W4)
else()
    target_compile_options(autosplatoon_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
qt_add_executable(AutoSplatoon WIN32 MACOSX_BUNDLE AutoSplatoon/main.cpp)
target_link_libraries(AutoSplatoon PRIVATE autosplatoon_core)
target_compile_definitions(AutoSplatoon PRIVATE AUTOSPLATOON_VERSION="${PROJECT_VERSION}")
set_target_properties(AutoSplatoon PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "org.autosplatoon.desktop"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
if(BUILD_TESTING)
    find_package(Qt6 ${qt_version} EXACT REQUIRED COMPONENTS Test)
    qt_add_executable(autosplatoon_tests tests/test_core.cpp)
    add_executable(fake_esptool tests/fake_esptool.cpp)
    target_link_libraries(fake_esptool PRIVATE Qt6::Core)
    add_dependencies(autosplatoon_tests fake_esptool)
    target_link_libraries(autosplatoon_tests PRIVATE autosplatoon_core Qt6::Test)
    add_test(NAME core COMMAND autosplatoon_tests)
    set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

set(AUTOSPLATOON_ASSET_DIR "${CMAKE_BINARY_DIR}/assets" CACHE PATH "Output of tools/fetch_assets.py")
set(tool_name esptool)
if(WIN32)
    set(tool_name esptool.exe)
endif()
install(TARGETS AutoSplatoon BUNDLE DESTINATION . RUNTIME DESTINATION .)
if(APPLE)
    set(asset_dest "AutoSplatoon.app/Contents/Resources")
else()
    set(asset_dest ".")
endif()
install(CODE "if(NOT EXISTS \"${AUTOSPLATOON_ASSET_DIR}/tools/${tool_name}\")\n message(FATAL_ERROR \"Missing bundled esptool. Run tools/fetch_assets.py first.\")\nendif()")
install(PROGRAMS "${AUTOSPLATOON_ASSET_DIR}/tools/${tool_name}" DESTINATION "${asset_dest}/tools")
install(DIRECTORY firmware licenses DESTINATION "${asset_dest}")
install(CODE "execute_process(COMMAND \"${CMAKE_COMMAND}\" --install \"${opencv_BINARY_DIR}\" --component licenses --prefix \"\${CMAKE_INSTALL_PREFIX}/${asset_dest}/licenses/OpenCV-third-party\" COMMAND_ERROR_IS_FATAL ANY)")
install(FILES LICENSE dependencies.lock.json DESTINATION "${asset_dest}")
install(FILES README.md DESTINATION "${asset_dest}")
install(DIRECTORY docs DESTINATION "${asset_dest}")
# Static OpenCV is built from source; exclude its SDK from application packages.
set(CPACK_INSTALL_CMAKE_PROJECTS "${CMAKE_BINARY_DIR};AutoSplatoon;Unspecified;/")
get_filename_component(qt_prefix "${Qt6_DIR}/../../.." ABSOLUTE)
set(deploy_options)
if(WIN32)
    # Widgets use raster rendering; software OpenGL and shader compilers are unused.
    set(deploy_options --no-opengl-sw --no-system-d3d-compiler --no-system-dxc-compiler)
elseif(APPLE)
    # macdeployqt only selects Cocoa by default; also relocate the QA platform plugin.
    install(FILES "${qt_prefix}/plugins/platforms/libqoffscreen.dylib"
        DESTINATION "AutoSplatoon.app/Contents/PlugIns/platforms")
    set(deploy_options "-executable=AutoSplatoon.app/Contents/PlugIns/platforms/libqoffscreen.dylib")
endif()
qt_generate_deploy_app_script(TARGET AutoSplatoon OUTPUT_SCRIPT qt_deploy NO_UNSUPPORTED_PLATFORM_ERROR
    EXCLUDE_PLUGIN_TYPES generic networkinformation tls INCLUDE_PLUGINS qoffscreen
    DEPLOY_TOOL_OPTIONS ${deploy_options})
install(SCRIPT ${qt_deploy})
install(FILES "${qt_prefix}/translations/qtbase_zh_CN.qm" DESTINATION "${asset_dest}/translations")
install(DIRECTORY "${qt_prefix}/sbom/" DESTINATION "${asset_dest}/licenses/Qt-SBOM")
if(MINGW)
    get_filename_component(compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    install(FILES "${compiler_bin}/libgcc_s_seh-1.dll" "${compiler_bin}/libstdc++-6.dll"
        "${compiler_bin}/libwinpthread-1.dll" DESTINATION .)
    install(DIRECTORY "${compiler_bin}/../licenses/gcc" "${compiler_bin}/../licenses/winpthreads"
        "${compiler_bin}/../licenses/mingw-w64" DESTINATION licenses/MinGW)
endif()
set(CPACK_PACKAGE_NAME AutoSplatoon)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "AutoSplatoon-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
if(APPLE)
    set(CPACK_GENERATOR DragNDrop)
elseif(WIN32)
    set(CPACK_GENERATOR ZIP)
else()
    set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
